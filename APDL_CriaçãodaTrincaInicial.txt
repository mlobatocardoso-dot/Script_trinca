
! ==========================================
! 1.0 ENTRAR NO PRÉ-PROCESSAMENTO
! ==========================================
/CLEAR, START          ! Limpa toda a memória do ANSYS e reseta o banco de dados
RESUME                 ! CARREGA o arquivo salvo (.db)
/PREP7                 ! Entra no módulo de modelagem e malha

! ==========================================
! 1.1 PARÂMETROS DE ENTRADA 
! ==========================================
desloc_z = -1.5e-3     ! Distância do offset do WorkPlane no eixo Z (anda no sentido negativo de z)
DistKP_YGlobal = 0.817e-2 ! Distancia entre keypoints na direção Y global (metade da espessura)

! Coordendas X e Z (no sist. global) do keypoint de referencia de onde se inicia a trinca A_TRINCA_SUP 
COORDX_KP_TRINCA = 5.245350
COORDZ_KP_TRINCA = 0.9442500 

! ==========================================
! 2 MODELAGEM DA TRINCA INICIAL
! ==========================================

WPSTYLE,,,,,,,,1       ! Ativa a visualização do WorkPlane na tela
WPOFF, 0, 0, desloc_z  ! Move o plano de trabalho no eixo Z Workplane.
VSBW, 10               ! Corta o Volume 10 usando o plano de trabalho
VSBW, 11               ! Corta o Volume 11 usando o plano de trabalho

! queremos excluir os volumes que estão na frente a uma distancia de  desloc_z 
CSYS, 4                ! Ativa o sistema de coordenadas do WorkPlane
VSEL, S, LOC, Z,0,-desloc_z    !VSEL, S, LOC, Z, VMIN, VMAX ! seleciona o pequeno trecho que foi criado com o corte
VDELE, ALL, , , 1              ! O '1' na posição correta vai pulverizar áreas, linhas e keypoints!

ALLSEL, ALL                    ! Seleciona tudo o que sobrou do modelo

!Aproveita para salvar o ID das áreas originais e que farão parte da trinca passante
ASEL, S, LOC, Z,desloc_z-1e-5,0+1e-5
*GET, A_ANTIGA_INF, AREA,, NUM, MAX
*GET, A_ANTIGA_SUP, AREA,, NUM, MIN

!Criar novos volumes pelo extrude das áreas 143 e 96 ( Na face imutável e original da extremidade da trinca. São áreas que fazem parte da componente VOL_SOLDA)
A_orig_sup = 143         ! Superior
A_orig_inf = 96          ! Inferior

! ------------------------------------------------------
! 2.1 REALIZA O EXTRUDE para criar os novos volumes E SALVA O ID dos volumes e áreas QUE SURGEM NA EXTREMIDADE E DE ONDE PARTIRÁ A O INICIO DA TRINCA
! 2.1.1: Isolar e calcular centroides (obs: eles são gerados pelo *get no sistema global)
! ------------------------------------------------------
ASEL, S,,, A_orig_sup
ASUM !O comando ASUM no ANSYS APDL calcula e exibe estatísticas geométricas para áreas selecionadas.
*GET, Z_centroide_sup, AREA, 0,CENT,Z
*GET, X_centroide_sup, AREA, 0,CENT,X
*GET, Y_centroide_sup, AREA, 0,CENT,Y

VSEL,NONE ! Esvazia a gaveta de volumes.
VOFFST, A_orig_sup, desloc_z, , !Realiza o extrude a partir da área A_orig_sup
*GET, V_NOVO_SUP, VOLU,, NUM, MAX

ASEL, S,,, A_orig_inf
ASUM
*GET, Z_centroide_inf, AREA, 0,CENT,Z
*GET, X_centroide_inf, AREA, 0,CENT,X
*GET, Y_centroide_inf, AREA, 0,CENT,Y

VSEL,NONE ! Esvazia a gaveta de volumes.
VOFFST, A_orig_inf, desloc_z, , 
*GET, V_NOVO_INF, VOLU,, NUM, MAX

! Passo 2.1.2 faz um merge entre o volume criado e a componente Volume de solda (criada previamente)
CMSEL, S, VOL_SOLDA, VOLU   

VSEL, A, , , V_NOVO_SUP !Adiciona (letra'A') os volumes V_NOVO_SUP e V_NOVO_INF à seleção atual  
VSEL, A, , , V_NOVO_INF 

ASLV, S  !Seleciona as áreas anexadas a ESSES volumes (mantendo os volumes selecionados)
LSLA, S  !Seleciona as linhas anexadas a ESSAS áreas (mantendo as áreas selecionadas)
KSLL, S  !Seleciona os keypoints anexados a ESSAS linhas (mantendo as linhas selecionadas)


NUMMRG, ALL, , , , LOW  ! Merge de tudo (Keypoints, Linhas, Áreas, Nós e Elementos) da região isolada 

ALLSEL, ALL
CSYS, 0                        ! Volta para o sistema Global

! ------------------------------------------------------
! 2.1.3: ! Calcula a coordenada Y esperada da extremidade 
! ------------------------------------------------------
! lembrar que Z_WP = Y_global e Y_WP = -Z_global

Y_novo_ext = Y_centroide_sup + desloc_z

! ------------------------------------------------------
!2.1.4: Filtrar e salvar as novas áreas (Superior e inferior) na extremidade do volume criado (A_EXTREMA), 
! objetivando os recuperar o ID dessas áreas
! ------------------------------------------------------

!2.1.5 Áreas dos volumes novos (superior e inferior)
VSEL, S,,, V_NOVO_SUP
ASLV, S
ASEL, R, LOC, Y, Y_novo_ext - 1E-5, Y_novo_ext + 1E-5
*GET, A_EXTREMA_SUP, AREA,, NUM, MAX

VSEL, S,,, V_NOVO_INF
ASLV, S
ASEL, R, LOC, Y, Y_novo_ext - 1E-5, Y_novo_ext + 1E-5
*GET, A_EXTREMA_INF, AREA,, NUM, MAX

! Retorna a seleção total para o modelo
ALLSEL, ALL

!2.2 Criar spline em arco com inclinação inicial

! 2.2.1 Realizar o processo para a trinca que inicia na área nova
! 2.2.1.1 Definir o ID dos keypoints de inicio da trinca
! Ponto conhecido (P1 - INÍCIO do arco) - keypoint 116
ASEL,S,,,A_EXTREMA_SUP
LSLA,S
KSLL, S 
KSEL, R, LOC, X,COORDX_KP_TRINCA-1E-5,COORDX_KP_TRINCA+1E-5
KSEL, R, LOC, Y,Y_novo_ext-1E-5,Y_novo_ext+1E-5
KSEL, R, LOC, Z,COORDZ_KP_TRINCA-1E-5,COORDZ_KP_TRINCA+1E-5
*GET, KP1_TRINCA, KP,, NUM, MAX

X1 = KX(KP1_TRINCA)  ! Captura a coordenada X do Keypoint KP1_TRINCA
Y1 = KY(KP1_TRINCA)  ! Captura a coordenada Y do Keypoint KP1_TRINCA
Z1 = KZ(KP1_TRINCA)  ! Captura a coordenada Z do Keypoint KP1_TRINCA

!2.2.1.2 Calcular os parâmetros geométricos do arco que servirão para as faces duplicadas da trinca

PI = ACOS(-1)
Comp_arco = 5e-3
Ang_abertura = 15 ! em graus
Ang_corda = 135 ! em graus

! Converter de graus para radianos

Ang_corda_rad = Ang_corda*PI/180
Ang_abertura_rad = Ang_abertura*PI/180

! Calculos

! Raio R, corda C e flecha
Raio = Comp_arco/Ang_abertura_rad
Corda = 2.0*Raio*SIN(Ang_abertura_rad/2.0)
Flecha = Raio*(1.0-COS(Ang_abertura_rad/2.0))

! Coordenada do ponto 3
X3 = X1
Y3 = Y1+Corda*COS(Ang_corda_rad)
Z3 = Z1+Corda*SIN(Ang_corda_rad)

! Coordenada do ponto 2
! Primeiro pegamos o meio da reta entre P1 e P3:
Y_meio_reta = (Y1+Y3)/2.0
Z_meio_reta = (Z1+Z3)/2.0

! Agora aplicamos a Flecha na direção PERPENDICULAR exata (+90°):
X2 = X1
Y2 = Y_meio_reta+Flecha*COS(Ang_corda_rad+PI/2.0)
Z2 = Z_meio_reta+Flecha*SIN(Ang_corda_rad+PI/2.0)

! Criação dos Keypoints 1002 e 1003
K, 1002, X2, Y2, Z2      ! Ponto 2 (Meio)
K, 1003, X3, Y3, Z3      ! Ponto 3 (Fim)

! Cria a spline que passa pelos 3 pontos
BSPLIN, KP1_TRINCA, 1002, 1003
TRINCA_SPLINE=_RETURN   ! Captura o número que da splyne ANSYS acabou de criar

! Cria um ponto deslocado APENAS no eixo X
Comp_guia = 4.75e-3
K,1004,X1+Comp_guia,Y1,Z1  
L,KP1_TRINCA,1004                       ! Linha reta perfeita no eixo X
NLIN_EIXO=_RETURN       ! Captura o número que o ANSYS escolheu para a linha guia

! ARRASTAR A SPLINE PELO EIXO

ADRAG,TRINCA_SPLINE,,,,,,NLIN_EIXO
A_TRINCA_SUP=_RETURN    !Captura o ID da área que acabou de ser criada
! Duplica a área da trinca superior
N = 3 ! tamanho do salto inicial na numeração da área 
AGEN,2,A_TRINCA_SUP,,,0,0,0,,,
A_TRINCA_SUP_DUP = A_TRINCA_SUP+N

! Reduz a quantidade de linhas!
! Faz o merge entre os keypoints da A_EXTREMA_SUP e da A_TRINCA_SUP
! seleciona as áreas superiores da trinca
ASEL, S,,,A_EXTREMA_SUP
ASEL, A,,,A_TRINCA_SUP
! seleciona as linhas dessas áreas
LSLA, S
! Dentre essas linhas, restringe para apenas as linhas na posição Y_P1, Z_P1 e X1+Comp_guia
LSEL, R, LOC, X,X1+Comp_guia-1E-4,X1+Comp_guia+1E-4
! Adiciona no grupo a outra linha que foi criada como guia 
LSEL, A,,,NLIN_EIXO
! seleciona os keypoints que pertencem a essas linhas
KSLL, S 
! realiza o merge
NUMMRG, KP, 1e-3 
ALLSEL,ALL

! Faz o merge entre os keypoints da A_ANTIGA_SUP e da A_TRINCA_SUP_DUP
ASEL, S,,,A_ANTIGA_SUP
ASEL, A,,,A_TRINCA_SUP_DUP
! seleciona as linhas dessas áreas
LSLA, S
! Dentre essas linhas, restringe para apenas as linhas na posição Y_P1, Z_P1 e X1+Comp_guia
LSEL, R, LOC, X,X1-1E-4,X1+Comp_guia+1E-4
! seleciona os keypoints que pertencem a essas linhas
KSLL, S 
! realiza o merge
NUMMRG, KP, 1e-3 
ALLSEL,ALL

! 2.2.2 Realizar o processo para a trinca que inicia na área antiga inferior
! 2.2.2.1 Definir o ID dos keypoints de inicio da trinca
ASEL, S,,, A_EXTREMA_INF,
LSLA,S
KSLL, S 
KSEL, R, LOC, X,X1+Comp_guia-1E-3,X1+Comp_guia+1E-3
KSEL, R, LOC, Y,Y_novo_ext-1E-5,Y_novo_ext+1E-5
KSEL, R, LOC, Z,COORDZ_KP_TRINCA-1E-5,COORDZ_KP_TRINCA+1E-5
*GET, KP1A_TRINCA, KP,, NUM, MAX

X1A = KX(KP1A_TRINCA)   ! Captura a coordenada X do Keypoint KP1A_TRINCA
Y1A  = KY(KP1A_TRINCA)  ! Captura a coordenada Y do Keypoint KP1A_TRINCA
Z1A  = KZ(KP1A_TRINCA)  ! Captura a coordenada Z do Keypoint KP1A_TRINCA

! 2.2.2.2 Calcular os parâmetros geométricos do arco que servirão para as faces duplicadas da trinca

! Coordenada do ponto 3
X3A = X1A
Y3A = Y1A+Corda*COS(Ang_corda_rad)
Z3A = Z1A+Corda*SIN(Ang_corda_rad)

! Coordenada do ponto 2
! Primeiro pegamos o meio da reta entre P1 e P3:
Y_meio_reta = (Y1A+Y3A)/2.0
Z_meio_reta = (Z1A+Z3A)/2.0

! Agora aplicamos a Flecha na direção PERPENDICULAR exata (+90°):
X2A = X1A
Y2A = Y_meio_reta+Flecha*COS(Ang_corda_rad+PI/2.0)
Z2A = Z_meio_reta+Flecha*SIN(Ang_corda_rad+PI/2.0)

! Criação dos Keypoints 1005 e 1006
K, 1005, X2A, Y2A, Z2A      ! Ponto 5 (Meio)
K, 1006, X3A, Y3A, Z3A      ! Ponto 6 (Fim)

! Cria a spline que passa pelos 3 pontos
BSPLIN, KP1A_TRINCA, 1005, 1006
TRINCA_SPLINE2=_RETURN   ! Captura o número que o ANSYS acabou de escolher

!Criar uma guia perfeita no eixo x
! SALVA AS COORDENADAS DO KP DE ONDE SURGE A GUIA (NLIN_EIXO)

! Cria um ponto deslocado APENAS no eixo X
K,1007,X1A+Comp_guia,Y1A,Z1A   
L,KP1A_TRINCA,1007                       ! Linha reta perfeita no eixo X
NLIN_EIXO2=_RETURN       ! Captura o número que o ANSYS escolheu para a reta

! ARRASTAR A SPLINE PELO EIXO

ADRAG,TRINCA_SPLINE2,,,,,,NLIN_EIXO2
A_TRINCA_INF=_RETURN    !Captura o ID da área que acabou de ser criada

! Duplica a área da trinca inferior
AGEN,2,A_TRINCA_INF,,,0,0,0,,, 
A_TRINCA_INF_DUP = A_TRINCA_INF+1

! Reduz a quantidade de linhas!
! Reduz a quantidade de linhas!
! Faz o merge entre os keypoints da A_EXTREMA_INF e da A_TRINCA_INF
! seleciona as áreas superiores da trinca
ASEL, S,,,A_EXTREMA_INF
ASEL, A,,,A_TRINCA_INF
! seleciona as linhas dessas áreas
LSLA, S
! Dentre essas linhas, restringe para apenas as linhas na posição Y_P1, Z_P1 e X1+Comp_guia
LSEL, R, LOC, X,X1A+Comp_guia-1E-4,X1A+Comp_guia+1E-4
! Adiciona no grupo a outra linha que foi criada como guia 
LSEL, A,,,NLIN_EIXO2
! seleciona os keypoints que pertencem a essas linhas
KSLL, S 
! realiza o merge
NUMMRG, KP, 1e-3 
ALLSEL,ALL

! Faz o merge entre os keypoints da A_ANTIGA_INF e da A_TRINCA_INF_DUP
ASEL, S,,,A_ANTIGA_INF
ASEL, A,,,A_TRINCA_INF_DUP
! seleciona as linhas dessas áreas
LSLA, S
! Dentre essas linhas, restringe para apenas as linhas na posição Y_P1, Z_P1 e X1+Comp_guia
LSEL, R, LOC, X,X1A-1E-4,X1A+Comp_guia+1E-4
! seleciona os keypoints que pertencem a essas linhas
KSLL, S 
! realiza o merge
NUMMRG, KP, 1e-3 
ALLSEL,ALL

! Realiza o merge entre as linhas na interface das áreas superior e Inferior
! Superior e inferior da trinca A_TRINCA
ASEL, S,,,A_TRINCA_SUP
ASEL, A,,,A_TRINCA_INF
LSLA, S
KSLL, S
NUMMRG, KP, 1e-3 
ALLSEL,ALL

! Superior e inferior da trinca A_TRINCA_ _DUP
ASEL, S,,,A_TRINCA_SUP_DUP
ASEL, A,,,A_TRINCA_INF_DUP
LSLA, S
KSLL, S
NUMMRG, KP, 1e-3 
ALLSEL,ALL

/REPLOT
APLOT
